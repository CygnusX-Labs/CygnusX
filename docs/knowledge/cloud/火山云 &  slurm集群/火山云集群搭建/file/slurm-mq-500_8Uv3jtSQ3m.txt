[cluster Slurm-MQ-500]
Category = Schedulers
Enabled = true
IconUrl = static/cloud/cluster/ui/ClusterIcon/slurm.png


    [[node defaults]]
    KeyPairName = $KeyPairName
    Zone = $Zone
    VPCId = $VPCId
    SubnetId = $SubnetId
    KeyPairLocation = ~/.ssh/vector.pem

        [[[configuration]]]
        slurm.version = $SlurmVersion  # slurm版本
        slurm.accounting.enabled = $SlurmAccountingEnabled    # 是否开启slurm accounting
        slurm.accounting.host = $SlurmAccountingHost            # MySQL/Mariadb host
        slurm.accounting.port = $SlurmAccountingPort            # MySQL/Mariadb port (默认3306)
        slurm.accounting.db = $SlurmAccountingDB              # MySQL/Mariadb database
        slurm.accounting.user = $SlurmAccountingUser          # MySQL/Mariadb user
        slurm.accounting.password = $SlurmAccountingPassword  # MySQL/Mariadb password

        #User Management
        user.domain_service = $UserDomainService
        user.ldap_service_URI = $LdapServiceURI
        user.ldap_service_DN = $LdapServiceDN
        user.ldap_service_binddn = $LdapServicebinddn
        user.ldap_service_bindpw = $LdapServicebindpw
        user.access_deny = $AccessDeny # 禁止集群普通用户登录计算节点

        [[[cluster-init slurm:default]]] # 所有节点都会使用slurm这个project的default spec作为cluster-init的一部分

        [[[configuration]]] #所有节点默认的共享目录挂载参数
        vector.mounts.nfs_shared.type = nfs
        vector.mounts.nfs_shared.mountpoint = /shared
        vector.mounts.nfs_shared.export_path = $NFSSharedExportPath
        vector.mounts.nfs_shared.address = $NFSAddress
        vector.mounts.nfs_shared.options = $NFSSharedMountOptions

        [[[configuration]]] #所有节点默认的调度目录挂载参数
        vector.mounts.nfs_sched.type = nfs
        vector.mounts.nfs_sched.mountpoint = /sched

    [[node scheduler]]
    MachineType = $SchedulerMachineType # 管理节点的实例类型
    ImageId = $SchedulerImageId     # 管理节点的镜像名
    CloudInit = $SchedulerCloudInit
    DataVolumes = $SchedulerDataVolumes

        [[[configuration]]]
        vector.mounts.nfs_sched.disabled = true # 是否禁用nfs_sched；是，即管理节点不尝试挂载/sched；该选项只在混合调度时有用。
        vector.mounts.nfs_shared.disabled = ${NFSType != "External"} # 是否禁用nfs_shared；如果NFSType是Builtin则禁用，代表管理节点不会尝试挂载外部的共享目录。否则则会集成上方定义的默认配置进行挂载
        # vector.volume.system.size = 40  #实例系统盘大小，如使用该参数，则按照参数值进行系统盘创建，如不使用，则按照镜像大小进行系统盘创建

        [[[cluster-init slurm:scheduler]]]

        [[[network-interface eth0]]]
        AssociatePublicIpAddress = $UsePublicNetwork # 是否给scheduler创建并关联公网IP

        [[[volume sched]]] # 挂载的作为调度目录的数据盘参数
        Size = 30
        SSD = True
        Mount = builtinsched
        Persistent = False # 实例停机后是否保留数据盘

        [[[volume shared]]] # 挂载的作为共享目录（包含家目录）的数据盘参数
        Size = ${ifThenElse(NFSType == "Builtin", FilesystemSize, 0)} # 如果NFSType是Builtin，则取用户提供的Size，否则取0，0 代表不创建
        SSD = True
        Mount = builtinshared
        Persistent = False # 实例停机后是否保留数据盘

        [[[configuration]]] # 默认调度数据盘的格式化参数
        vector.mounts.builtinsched.mountpoint = /sched
        vector.mounts.builtinsched.fs_type = xfs

        [[[configuration]]] # 默认共享数据盘的格式化参数
        vector.mounts.builtinshared.disabled = ${NFSType != "Builtin"}
        vector.mounts.builtinshared.mountpoint = /shared
        vector.mounts.builtinshared.fs_type = xfs

        [[[configuration]]] # 默认调度目录的export参数
        vector.exports.builtinsched.export_path = /sched               # 调度目录强制挂载到/sched路径下
        vector.exports.builtinsched.options = no_root_squash
        vector.exports.builtinsched.type = nfs

        [[[configuration]]] # 默认共享目录的export参数
        vector.exports.builtinshared.disabled = ${NFSType != "Builtin"} # 如果NFSType不是builtin的，代表要挂载外部nas，此时无需从管理节点export共享目录
        vector.exports.builtinshared.export_path = /shared              # 共享目录强制挂载到/shared路径下
        vector.exports.builtinshared.type = nfs

    [[node login]]
    InitialCount = $NumberLoginNodes
    MachineType = $LoginMachineType
    ImageId = $LoginImageId
    CloudInit = $LoginCloudInit
    DataVolumes = $LoginDataVolumes

        [[[network-interface eth0]]]
        AssociatePublicIpAddress = $LoginNodesPublic # 是否为节点申请公网IP

        [[[configuration]]]
        # vector.volume.system.size = 40  #实例系统盘大小，如使用该参数，则按照参数值进行系统盘创建，如不使用，则按照镜像大小进行系统盘创建


    [[nodearray cpu1]]
    MachineType = $CPU1MachineType # CPU队列的实例类型
    ImageId = $CPU1ImageId     # CPU队列的镜像名
    MaxNodeCount = $MaxCPU1ExecuteNodeCount #最大节点数
    CloudInit = $CPU1CloudInit
    SubnetIds = $CPU1SubNetIds
    DataVolumes = $CPU1DataVolumes

        [[[configuration]]]
        slurm.autoscale = $Autoscale            # 是否弹性伸缩
        slurm.default_partition = true    # 是否是默认队列
        # vector.volume.system.size = 40  #实例系统盘大小，如使用该参数，则按照参数值进行系统盘创建，如不使用，则按照镜像大小进行系统盘创建

        [[[cluster-init slurm:execute]]]

        [[[network-interface eth0]]]
        AssociatePublicIpAddress = $ExecuteNodesPublic # 是否为节点申请公网IP

    [[nodearray cpu2]]
    MachineType = $CPU2MachineType # CPU队列的实例类型
    ImageId = $CPU2ImageId     # CPU队列的镜像名
    MaxNodeCount = $MaxCPU2ExecuteNodeCount #最大节点数
    CloudInit = $CPU2CloudInit

    DataVolumes = $CPU2DataVolumes

        [[[configuration]]]
        slurm.autoscale = $Autoscale            # 是否弹性伸缩
        slurm.default_partition = true    # 是否是默认队列
        # vector.volume.system.size = 40  #实例系统盘大小，如使用该参数，则按照参数值进行系统盘创建，如不使用，则按照镜像大小进行系统盘创建

        [[[cluster-init slurm:execute]]]

        [[[network-interface eth0]]]
        AssociatePublicIpAddress = $ExecuteNodesPublic # 是否为节点申请公网IP


    [[nodearray gpu]]
    MachineType = $GPUMachineType
    ImageId = $GPUImageId
    MaxNodeCount = $MaxGPUExecuteNodeCount
    CloudInit = $GPUCloudInit
    SubnetIds = $GPUSubNetIds
